# Build synthetic paired test set for PSNR/SSIM evaluation
# Clean source: RTTS/LFW preprocessed test images
# Degrade them synthetically so we have ground truth for PSNR/SSIM
import albumentations as A
from skimage.metrics import peak_signal_noise_ratio as compute_psnr
from skimage.metrics import structural_similarity as compute_ssim

DATASETS_DIR = '/content/computer_vision_expirement/datasets'

clean_sources = []
for ds in ['rtts_processed', 'lfw_processed', 'widerface_processed']:
    imgs = sorted(glob.glob(f'{DATASETS_DIR}/{ds}/test/*.jpg'))[:20]
    clean_sources.extend(imgs)
    if len(clean_sources) >= 20:
        break

deg_transforms = {
    'fog': A.Compose([A.RandomFog(fog_coef_lower=0.5, fog_coef_upper=0.7, alpha_coef=0.08, p=1.0)]),
    'low_light': A.Compose([A.RandomBrightnessContrast(
                     brightness_limit=(-0.5, -0.3), contrast_limit=(-0.3, 0.0), p=1.0)]),
    'rain': A.Compose([A.RandomRain(slant_lower=-10, slant_upper=10, drop_length=20,
                        drop_width=1, drop_color=(200, 200, 200),
                        blur_value=3, brightness_coefficient=0.7, p=1.0)]),
}

synthetic_pairs = []  # list of (clean_bgr, degraded_bgr, condition)
for img_path in clean_sources[:20]:
    img = cv2.imread(img_path)
    if img is None:
        continue
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    for cond, transform in deg_transforms.items():
        deg_rgb = transform(image=img_rgb)['image']
        deg_bgr = cv2.cvtColor(deg_rgb, cv2.COLOR_RGB2BGR)
        synthetic_pairs.append((img, deg_bgr, cond))

print(f'Synthetic paired images: {len(synthetic_pairs)}')
print(f'  ({len(clean_sources)} clean sources x {len(deg_transforms)} conditions)')
print('PSNR/SSIM will compare enhanced output against the clean original.')
