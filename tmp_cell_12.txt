import pyiqa
import time
import numpy as np
import pandas as pd
from skimage.metrics import peak_signal_noise_ratio as compute_psnr
from skimage.metrics import structural_similarity as compute_ssim

device = 'cuda' if torch.cuda.is_available() else 'cpu'
niqe_metric = pyiqa.create_metric('niqe', device=device)

def evaluate_no_ref(img_bgr):
    """NIQE (no-reference): lower = better."""
    t = torch.from_numpy(cv2.cvtColor(img_bgr, cv2.COLOR_BGR2RGB)).permute(2, 0, 1).unsqueeze(0).float() / 255.0
    return niqe_metric(t.to(device)).item()

def evaluate_ref(enhanced_bgr, clean_bgr):
    """PSNR and SSIM against clean reference (higher = better)."""
    enh = cv2.cvtColor(enhanced_bgr, cv2.COLOR_BGR2RGB).astype(np.float32)
    ref = cv2.cvtColor(clean_bgr,    cv2.COLOR_BGR2RGB).astype(np.float32)
    if enh.shape != ref.shape:
        enh = cv2.resize(cv2.cvtColor(enhanced_bgr, cv2.COLOR_BGR2RGB),
                         (ref.shape[1], ref.shape[0])).astype(np.float32)
    p = compute_psnr(ref, enh, data_range=255)
    s = compute_ssim(ref, enh, channel_axis=2, data_range=255)
    return p, s

    'FFA-Net':    enhance_ffanet,
}

all_results = {name: {'psnr': [], 'ssim': [], 'niqe': [], 'niqe_delta': [], 'latency': []}
               for name in models}

for clean_bgr, deg_bgr, cond in synthetic_pairs[:n_pairs]:
    for name, enhance_fn in models.items():
        try:
            start = time.time()
            enhanced = enhance_fn(deg_bgr)
            lat = (time.time() - start) * 1000
            if enhanced.shape[:2] != clean_bgr.shape[:2]:
                enhanced = cv2.resize(enhanced, (clean_bgr.shape[1], clean_bgr.shape[0]))
            p, s = evaluate_ref(enhanced, clean_bgr)
            all_results[name]['psnr'].append(p)
            all_results[name]['ssim'].append(s)
            all_results[name]['latency'].append(lat)
        except Exception as e:
            pass

    for name, enhance_fn in models.items():

            enhanced = enhance_fn(img)
            if enhanced.shape[:2] != img.shape[:2]:
                enhanced = cv2.resize(enhanced, (img.shape[1], img.shape[0]))
            enh_niqe = evaluate_no_ref(enhanced)
            all_results[name]['niqe'].append(enh_niqe)
            all_results[name]['niqe_delta'].append(enh_niqe - orig_niqe)

            pass
