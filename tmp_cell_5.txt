import cv2
import numpy as np
import glob
import os
import time
from pathlib import Path

def load_dataset_images(dataset_dir, dataset_name, split='test', max_samples=200):
    """Load images from a specific dataset split."""
    images = []
    search_dirs = [
        f'{dataset_dir}/{dataset_name}_processed/{split}',
        f'{dataset_dir}/{dataset_name}/{split}',
        f'{dataset_dir}/{dataset_name}',
    ]
    for search_dir in search_dirs:
        paths = sorted(
            glob.glob(f'{search_dir}/**/*.jpg', recursive=True) +
            glob.glob(f'{search_dir}/**/*.png', recursive=True) +
            glob.glob(f'{search_dir}/**/*.JPEG', recursive=True) +
            glob.glob(f'{search_dir}/**/*.JPG', recursive=True)
        )
        if paths:
            for p in paths[:max_samples]:
                img = cv2.imread(p)
                if img is not None:
                    images.append((p, img))
            break
    return images

# ── Step 2: Kaggle fallback (auth is set up by this point in the notebook) ──
import subprocess, sys as _sys

def kaggle_download(dataset_id, dest_dir, label):
    """Download a Kaggle dataset and return image count."""
    os.makedirs(dest_dir, exist_ok=True)
    print(f"  Downloading {label} from Kaggle ({dataset_id})...")
    _sys.stdout.flush()
    r = subprocess.run(
        ['kaggle', 'datasets', 'download', '-d', dataset_id,
         '-p', dest_dir, '--unzip'],
        capture_output=True, text=True, timeout=600
    )
    if r.returncode != 0:
        print(f"  Kaggle failed: {r.stderr[-300:]}")
        return 0
    count = sum(len(glob.glob(f'{dest_dir}/**/*.{ext}', recursive=True))
                for ext in ['jpg', 'png', 'JPEG', 'JPG', 'PNG'])
    print(f"  {label}: {count} images")
    return count

# ── Step 4: Synthetic degradation — guaranteed last resort ──
if not exdark_images or not foggy_images:
    print("[FALLBACK] Synthesising degraded images from any images on Drive...")
    any_imgs = []
    for ext in ['jpg', 'png', 'JPG', 'PNG']:
        any_imgs += glob.glob(f'{PROJECT_DIR}/**/*.{ext}', recursive=True)
    for p in any_imgs[:50]:
        img = cv2.imread(p)
        if img is None:
            continue
        if not exdark_images:
            dark = (((img.astype('float32') / 255.0) ** 2.5) * 255).clip(0, 255).astype('uint8')
            exdark_images.append((p + '_synth_dark', dark))
        if not foggy_images:
            hazy = cv2.addWeighted(img, 0.4, (np.ones_like(img) * 220).astype('uint8'), 0.6, 0)
            foggy_images.append((p + '_synth_haze', hazy))
    if exdark_images:
        print(f"  Synthetic low-light images created: {len(exdark_images)}")
    if foggy_images:
        print(f"  Synthetic hazy images created: {len(foggy_images)}")
