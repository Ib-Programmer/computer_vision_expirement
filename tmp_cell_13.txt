# Save enhanced outputs for each model to Drive (deliverable 2.2)
import os

ENHANCED_DIR = f'{RESULTS_DIR}/enhanced_outputs'
os.makedirs(ENHANCED_DIR, exist_ok=True)

n_save = min(20, len(all_test_images))
saved = {name: 0 for name in models}

for name, enhance_fn in models.items():
    model_dir = f'{ENHANCED_DIR}/{name.replace("+","plus").replace(" ","_")}'
    os.makedirs(model_dir, exist_ok=True)
    for idx, (path, img) in enumerate(all_test_images[:n_save]):
        try:
            enhanced = enhance_fn(img)
            cv2.imwrite(f'{model_dir}/{os.path.basename(path)}', enhanced)
            saved[name] += 1
        except Exception:
            pass

print('Enhanced outputs saved to Drive:')
for name, n in saved.items():
    print(f'  {name}: {n} images -> {model_dir}')
